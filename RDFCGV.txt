4.	Design, Develop and Implement a menu driven Program in C for the following operations on Singly Linked List (SLL) of Student Data with the fields: USN, Name, Branch, Sem, PhNo.
#include <stdio.h>
#include <stdlib.h>

struct student {
    char usn[20], name[30], branch[20], phno[15];
    int sem;
    struct student *next;
};

struct student *head = NULL;

struct student *getnode() {
    struct student *p = malloc(sizeof(struct student));

    printf("Enter USN Name Branch Sem Phone:\n");
    scanf("%19s %29s %19s %d %14s",
          p->usn, p->name, p->branch, &p->sem, p->phno);

    p->next = NULL;
    return p;
}

void front() {
    struct student *p = getnode();
    p->next = head;
    head = p;
}

void end() {
    struct student *p = getnode(), *t = head;

    if (head == NULL)
        head = p;
    else {
        while (t->next != NULL)
            t = t->next;
        t->next = p;
    }
}

void deleteFront() {
    struct student *t = head;

    if (t == NULL) {
        printf("List empty\n");
        return;
    }

    head = t->next;
    free(t);
}

void deleteEnd() {
    struct student *t = head, *prev = NULL;

    if (t == NULL) {
        printf("List empty\n");
        return;
    }

    while (t->next != NULL) {
        prev = t;
        t = t->next;
    }

    if (prev == NULL)
        head = NULL;
    else
        prev->next = NULL;

    free(t);
}

void display() {
    struct student *t = head;
    int count = 0;

    while (t != NULL) {
        printf("\n%s %s %s %d %s\n",
               t->usn, t->name, t->branch, t->sem, t->phno);
        count++;
        t = t->next;
    }

    printf("Number of nodes = %d\n", count);
}

int main() {
    int ch, n, i;

    while (1) {
        printf("\n1.Create  2.Display  3.Insert End");
        printf("\n4.Delete End  5.Insert Front");
        printf("\n6.Delete Front  7.Exit\n");

        scanf("%d", &ch);

        switch (ch) {
            case 1:
                printf("Enter N: ");
                scanf("%d", &n);
                for (i = 0; i < n; i++)
                    front();
                break;
            case 2: display(); break;
            case 3: end(); break;
            case 4: deleteEnd(); break;
            case 5: front(); break;
            case 6: deleteFront(); break;
            case 7: return 0;
            default: printf("Invalid choice\n");
        }
    }
}

5.	Design, Develop and Implement a menu driven Program in C for the following operations on Doubly Linked List (DLL).
#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *prev, *next;
} *head = NULL;

struct node *getnode() {
    struct node *p = malloc(sizeof(struct node));
    printf("Enter data: ");
    scanf("%d", &p->data);
    p->prev = p->next = NULL;
    return p;
}

void front() {
    struct node *p = getnode();
    p->next = head;
    if (head) head->prev = p;
    head = p;
}

void end() {
    struct node *p = getnode(), *t = head;
    if (!head) head = p;
    else {
        while (t->next) t = t->next;
        t->next = p;
        p->prev = t;
    }
}

void delFront() {
    if (!head) return;
    struct node *t = head;
    head = head->next;
    if (head) head->prev = NULL;
    free(t);
}

void delEnd() {
    if (!head) return;
    struct node *t = head;
    while (t->next) t = t->next;
    if (t->prev) t->prev->next = NULL;
    else head = NULL;
    free(t);
}

void display() {
    struct node *t = head;
    int count = 0;
    while (t) {
        printf("%d <-> ", t->data);
        count++;
        t = t->next;
    }
    printf("NULL\nNodes = %d\n", count);
}

int main() {
    int ch, n, i;
    while (1) {
        printf("\n1.Create 2.Display 3.InsertEnd 4.DeleteEnd");
        printf("\n5.InsertFront 6.DeleteFront 7.Exit\n");
        scanf("%d", &ch);

        switch (ch) {
            case 1:
                printf("Enter N: ");
                scanf("%d", &n);
                for (i = 0; i < n; i++) end();
                break;
            case 2: display(); break;
            case 3: end(); break;
            case 4: delEnd(); break;
            case 5: front(); break;
            case 6: delFront(); break;
            case 7: return 0;
        }
    }
}
3.	Design, Develop and Implement a menu driven Program in C for the following operations on Circular QUEUE of Characters

#include <stdio.h>
#define MAX 5

int a[MAX], f = 0, r = -1, Count = 0;

void display() {
    int i, j;

    if (Count == 0) {
        printf("Queue is Empty\n");
        return;
    }

    i = f;
    for (j = 1; j <= Count; j++) {
        printf("%d\t", a[i]);
        i = (i + 1) % MAX;
    }
    printf("\n");
}

void insert(int item) {
    if (Count == MAX) {
        printf("Queue is Full\n");
        return;
    }

    r = (r + 1) % MAX;
    a[r] = item;
    Count++;
}

int delet() {
    int itemdel;

    if (Count == 0) {
        printf("Queue is Empty\n");
        return 0;
    }

    itemdel = a[f];
    f = (f + 1) % MAX;
    Count--;

    return itemdel;
}

int main() {
    int ch, item, itemdel;

    while (1) {
        printf("\n1. Insert");
        printf("\n2. Delete");
        printf("\n3. Display");
        printf("\n4. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        switch (ch) {
            case 1:
                printf("Enter the item: ");
                scanf("%d", &item);
                insert(item);
                break;

            case 2:
                itemdel = delet();
                if (Count >= 0)
                    printf("Deleted item: %d\n", itemdel);
                break;

            case 3:
                display();
                break;

            case 4:
                return 0;

            default:
                printf("Invalid choice\n");
        }
    }
}
Program 6: Maximum Height and Level of a Binary Tree
#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *left, *right;
};

int maxDepth(struct node *root) {
    if (root == NULL)
        return 0;

    int l = maxDepth(root->left);
    int r = maxDepth(root->right);

    return (l > r ? l : r) + 1;
}

struct node* newNode(int data) {
    struct node *n = malloc(sizeof(struct node));
    n->data = data;
    n->left = n->right = NULL;
    return n;
}

int getLevel(struct node *root, int data, int level) {
    if (root == NULL)
        return 0;

    if (root->data == data)
        return level;

    int l = getLevel(root->left, data, level + 1);

    if (l != 0)
        return l;

    return getLevel(root->right, data, level + 1);
}

int main() {
    struct node *root = newNode(3);
    root->left = newNode(2);
    root->right = newNode(5);
    root->left->left = newNode(1);
    root->left->right = newNode(4);

    printf("Height of tree is %d\n", maxDepth(root));

    for (int x = 1; x <= 5; x++)
        printf("Level of %d is %d\n", x,
               getLevel(root, x, 1));

    return 0;
}
Program 7: Print All Root-to-Leaf Paths in a Binary Tree

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *left, *right;
};

struct node* newnode(int data) {
    struct node *n = malloc(sizeof(struct node));
    n->data = data;
    n->left = n->right = NULL;
    return n;
}

void printPaths(struct node *root, int path[], int len) {
    if (root == NULL)
        return;

    path[len++] = root->data;

    if (root->left == NULL && root->right == NULL) {
        for (int i = 0; i < len; i++)
            printf("%d ", path[i]);
        printf("\n");
    } else {
        printPaths(root->left, path, len);
        printPaths(root->right, path, len);
    }
}

int main() {
    int path[1000];

    struct node *root = newnode(10);
    root->left = newnode(8);
    root->right = newnode(2);
    root->left->left = newnode(3);
    root->left->right = newnode(5);
    root->right->left = newnode(2);

    printPaths(root, path, 0);

    return 0;
}
